CREATE TABLE worker_info (
    worker_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    first_name TEXT,
    middle_name TEXT,
    last_name TEXT,
    date_of_birth DATE,
    address TEXT,
    city TEXT,
    state TEXT,
    country TEXT,
    postal_code TEXT,
    phone_number TEXT,
    alternate_phone_number TEXT,
    email TEXT,
    gender_id INTEGER,
    nationality TEXT
);

CREATE TABLE professional_info (
    professional_id INTEGER GENERATED BY DEFAULT AS IDENTITY,
    worker_id INTEGER,
    qualification TEXT,
    specialization TEXT,
    years_of_experience INTEGER,
    previous_company TEXT,
    previous_job_title TEXT,
    previous_job_start_date DATE,
    previous_job_end_date DATE,
    FOREIGN KEY (worker_id) REFERENCES worker_info(worker_id)
);

CREATE TABLE worker_jobs (
    job_id INTEGER GENERATED BY DEFAULT AS IDENTITY,
    worker_id INTEGER,
    job_start_date DATE,
    job_end_date DATE,
    salary NUMERIC(10,2),
    contract_type_id INTEGER,
    department TEXT,
    FOREIGN KEY (worker_id) REFERENCES worker_info(worker_id)
);
